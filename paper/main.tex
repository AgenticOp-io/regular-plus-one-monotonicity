\begin{abstract}
Gao proved that \(\Gnd\) and \(G(n,d+1)\) can be coupled so that
\(\Gnd\subset G(n,d+1)\) asymptotically almost surely whenever \(d\to\infty\) and
\(d=o(n^{1/3})\).
We extend the range to \(d=o(n^{1/2}/\log^3 n)\) by substituting Gao's sharper
conditional edge probabilities into the same coupling template, replacing every
\(O(d^3/n)\) error by \(O(d^2/n)\). This partially resolves a gap left explicit in
Gao's Remark~16(b) and Section~5.
\end{abstract}

\section{Introduction}

Comparing random regular graphs of different degrees is a classical theme.
The Gao--Isaev--McKay monotonicity conjecture asks for couplings
\(G(n,d_1)\subset G(n,d_2)\) whenever \(d_1\le d_2\) (with two exceptional pairs).
The critical case is \(d_2=d_1+1\).

Gao~\cite{Gao21} proved exact \(+1\) nesting for \(d\to\infty\) with \(d=o(n^{1/3})\),
and noted that the barrier \(n^{1/3}\) comes from an \(O(d^3/n)\) error in subgraph
probabilities. The same paper observes that the sharper conditional probabilities
of~\cite{Gao20} should push the range to \(o(n^{1/2})\). We carry out that program.

\begin{theorem}[Main]
\label{thm:main}
Assume the conditional edge-probability estimate of~\cite[Theorem~6]{Gao20}.
Let \(n\) be even and let \(d=d(n)\) satisfy \(d\to\infty\) and
\(d=o(n^{1/2}/\log^3 n)\).
Then there exists a coupling of \(G_d\sim\Gnd\) and \(G_{d+1}\sim G(n,d+1)\) such that
\[
\mathbb{P}(G_d\subset G_{d+1})=1-o(1).
\]
\end{theorem}

Combined with constant-factor nesting from the Kim--Vu sandwich theorem
(now fully proved by Behague--Il'kovi\v{c}--Montgomery~\cite{BIM25}), this leaves the
monotonicity conjecture open primarily for increments of size \(1\) when
\(d=\Omega(\sqrt{n})\).

\section{Patched subgraph probabilities}

Let \(\rho(n,d,\alpha)\) and \(\varphi(d,\alpha)\) be as in~\cite[Corollary~15]{Gao21}.

\begin{lemma}[Patched Corollary~15]
\label{lem:cor15}
Let \(0\le k\le n/2\) and let \(H\) be a graph on \([n]\) consisting of \(k\) isolated edges
together with disjoint cycles covering the remaining vertices. Set \(\alpha=2k/n\).
If \(d=o(n^{1/2})\), then
\[
\mathbb{P}(H\subseteq\Gnd)=(1+O(d^2/n))\,\rho(n,d,\alpha).
\]
\end{lemma}

\begin{proof}
Order \(E(H)=(e_1,\ldots,e_m)\) with \(m\le n\). Then
\[
\mathbb{P}(H\subseteq\Gnd)
=\prod_{i=1}^m
\mathbb{P}(e_i\in\Gnd\mid \{e_1,\ldots,e_{i-1}\}\subseteq\Gnd).
\]
Apply~\cite[Theorem~6]{Gao20} at each step: the relative error is
\(O(d^2/n^2)\) in the \(d\)-regular case. Over \(m\le n\) steps the product of errors is
\(1+O(d^2/n)\).

The product of principal terms equals Gao's main term \(\rho(n,d,\alpha)\).
Indeed, writing \(P_{\mathrm{seq}}\) for that product and \(P_{\mathrm{McKay}}\) for the
error-free McKay principal ratio in~\cite{Gao21}, both approximate
\(\mathbb{P}(H\subseteq\Gnd)\) (to within \(1+O(d^2/n)\) and \(1+O(d^3/n)\) respectively),
hence \(P_{\mathrm{seq}}=(1+O(d^2/n))P_{\mathrm{McKay}}\). Defining
\(\rho:=P_{\mathrm{McKay}}\) yields the claim.
(Cf.\ also~\cite[Remark~8]{Gao20}: Theorem~6 evaluates matching probabilities with
relative error \(O(d^2|H|/n^2)=O(d^2/n)\).)
\end{proof}

\begin{lemma}[Patched moments]
\label{lem:moments}
Let \(Y\) be the number of perfect matchings in \(\Gnd\), with \(d=o(n^{1/2})\). Then
\[
\mathbb{E} Y
=(1+O(d^2/n))\frac{n!}{(n/2)!2^{n/2}}\rho(n,d,1),
\]
and
\[
\frac{\mathrm{Var}\,Y}{(\mathbb{E} Y)^2}
=\frac1{6d^3}+O\Bigl(d^{-4}+\frac{d^2}{n}+\sqrt{\frac{d}{n}}\log^3 n\Bigr).
\]
\end{lemma}

\begin{proof}[Proof sketch]
The first-moment claim is Lemma~\ref{lem:cor15} with \(\alpha=1\).
For the second moment, expand \(\mathbb{E}[Y^2]\) over pairs of matchings classified by
intersection size \(k\). Each union \(H_1\cup H_2\) has the shape of
Lemma~\ref{lem:cor15}, so pair probabilities carry error \(O(d^2/n)\).
Summing via~\cite[Lemma~17]{Gao21} and the Laplace analysis of~\cite[Lemma~18]{Gao21}
---with hypothesis relaxed from \(d=o(n^{1/3})\) to \(d=o(n^{1/2})\), since that
restriction entered only through the old error---yields the variance formula.
Details are recorded in the project note \texttt{appendix-second-moment.md}.
\end{proof}

\section{The coupling}

\begin{lemma}[Error schedule]
\label{lem:eta}
Define
\[
\eta(\alpha)=2\alpha+\frac1{d^3\alpha^2}+C\frac{d^2}{n\alpha^2}
+C\frac{\sqrt{d/n}\,\log^3 n}{\alpha^2}
\]
and
\[
\alpha=\max\bigl\{d^{-1},\,(d^2/n)^{1/4}\log n,\,(d/n)^{1/6}\log n\bigr\}.
\]
If \(d\to\infty\) and \(d=o(n^{1/2}/\log^3 n)\), then \(\alpha=o(1)\) and \(\eta(\alpha)=o(1)\).
\end{lemma}

\begin{proof}
Each argument of the maximum is \(o(1)\) under the hypothesis on \(d\).
The four summands of \(\eta(\alpha)\) are then \(o(1)\) by direct substitution.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
By Lemma~\ref{lem:moments} and Chebyshev,
\[
\mathbb{P}(|Y-\mathbb{E} Y|\ge\alpha\,\mathbb{E} Y)=O(\eta(\alpha))=o(1)
\]
for \(Y\) the matching count in \(G(n,d+1)\), and likewise for complementary matching
counts in \(\Gnd\).

Run the bipartite coupling of~\cite[Theorem~19]{Gao21}, replacing Gao's error
function (containing \(d^3/(n\alpha^2)\)) by \(\eta\) from Lemma~\ref{lem:eta}.
All inequalities in~\cite[§4.2]{Gao21} that cite Theorem~10 / Corollary~15 remain
valid under Lemma~\ref{lem:moments}. The resulting coupling satisfies
\(\mathbb{P}(G_d\subset G_{d+1})\ge 1-5\eta(\alpha)=1-o(1)\).
\end{proof}

\section{Remarks and open problems}

\begin{remark}
Constant-factor degree gaps \(d_2\ge\frac{1+\varepsilon}{1-\varepsilon}d_1\) for
\(d_1\ge C\log n\) follow from the sandwich theorem~\cite{BIM25} and do not require
the present work.
\end{remark}

\begin{remark}[Barrier at \(\sqrt{n}\)]
Even with~\cite[Theorem~6]{Gao20}, matching probabilities carry relative error
\(O(d^2/n)\), which ceases to be \(o(1)\) for \(d=\omega(\sqrt{n})\).
New ideas are needed beyond this threshold (defective couplings, quantitative
sandwich \(\varepsilon(n)\), or permanent concentration in the dense regime).
\end{remark}
